Build production-shaped AI agents with the Lyzr Agent Development Kit — the Python module we use to ship agents ourselves. Free. Self-paced.

from lyzr import Agent

agent = Agent(
  name="researcher",
  llm="openai/gpt-5",
  memory=True,
  tools=[web_search, calculator],
)

response = agent.run(
  "what changed in agent
   building this quarter?"
)

for chunk in response.stream():
  print(chunk, end="")
